+module RfqProcessing
+  # PDF schedules carry stronger item boundaries than numbered legal paragraphs.
+  # Prefer an explicit unit/quantity table; retain source words, including empty
+  # price cells, without manufacturing a price or a demand from packaging.
+  module ItemLines
+    UNIT = /(?:kg|kilograms?|lbs?|pounds?|ea|each|pcs?|pieces?|sheets?|plates?|bars?|bags?|boxes|tons?|tonnes?|metres?|meters?|m|ft)/i
+    NUMBER = /\d+(?:[ ,]\d{3})*(?:\.\d+)?/
+    PRODUCT = /\b(?:steel|stainless|alum(?:inium|inum)?|galv(?:anized)?|sheet|plate|bar|tube|pipe|angle|channel|bolt|shot|grit|ingot|garnet|beads?|AN\d+)\b/i
+    BOILERPLATE = /\b(?:bidder|respondent|quotation no|request for quotation|closing date|tender|tax|vat|rand value|estimated value|total price|total inclusive|discount|signature|purchase order|delivery lead.time|terms and conditions)\b/i
+    TABLE_END = /\b(?:total|subtotal|vat|discount|signature|delivery lead.time|respondents?|terms|section)\b/i
+    HEADER = /\b(?:unit|uom)\b.*\b(?:quantity|qty)\b|\b(?:quantity|qty)\b.*\b(?:unit|uom)\b/i
+    DEMAND = /\b#{UNIT}\s+#{NUMBER}\b|\b#{NUMBER}\s*#{UNIT}\b/i
+
+    def self.pdf(extraction)
+      table_rows = []
+      in_table = false
+      number_column = false
+      current = []
+      page = nil
+      extraction.lines.each do |line|
+        line_page = line.location[/\Apage \d+/]
+        if line_page != page
+          unless current.empty?
+            extraction.warnings << "an item table reaches the end of #{page}; review the next page for continuation"
+            table_rows << assemble(current)
+          end
+          current = []
+          in_table = false
+          page = line_page
+        end
+        if HEADER.match?(line.text)
+          table_rows << assemble(current) unless current.empty?
+          current = []
+          in_table = true
+          number_column = line.text.match?(/\Aline\b/i)
+          next
+        end
+        next unless in_table
+
+        if TABLE_END.match?(line.text)
+          table_rows << assemble(current) unless current.empty?
+          current = []
+          in_table = false
+        elsif table_row?(line)
+          table_rows << assemble(current) unless current.empty?
+          current = [number_column ? without_line_number(line) : line]
+        elsif !current.empty?
+          # Wrapped description cells stay within the description column. Reject
+          # left-margin paragraphs and standalone monetary/form fields.
+          raw = line.layout_text || line.text
+          previous = current.first.layout_text || current.first.text
+          if raw.index(/\S/).to_i > previous.index(/\S/).to_i && !BOILERPLATE.match?(line.text)
+            current << line
+          else
+            table_rows << assemble(current)
+            current = []
+            in_table = false
+          end
+        end
+      end
+      table_rows << assemble(current) unless current.empty?
+      return table_rows unless table_rows.empty?
+
+      extraction.warnings << "no quantity table was recognized in the PDF; review the source for omitted or wrapped items"
+      # Untabulated PDFs need both a physical product and a measurable demand,
+      # dimension, or part designation. Page numbers and financial thresholds do
+      # not become quote lines merely because they contain digits.
+      extraction.lines.select do |line|
+        text = line.text
+        PRODUCT.match?(text) && !BOILERPLATE.match?(text) &&
+          !text.match?(/\b(?:container|pack(?:age|aging)?|per box)\b/i) &&
+          (DEMAND.match?(text) || text.match?(/\d\s*(?:x|×|ga\b|gauge|mm\b|in\b|"|')|\bAN\d+/i))
+      end
+    end
+
+    # Carry explicit body requirements into compliance, not the entire tender's
+    # legal boilerplate or the notes belonging to another schedule row.
+    def self.body_text(extraction)
+      # Reuse the same boundaries that assembled wrapped descriptions. A note
+      # inside an item's evidence span belongs to that item, never every row.
+      # Selection here must not duplicate warnings already produced by pdf.
+      selection = extraction.dup
+      selection.warnings = []
+      item_spans = pdf(selection).filter_map do |item|
+        match = /\Apage (\d+), lines? (\d+)(?:-(\d+))?\z/.match(item.location)
+        [match[1], (match[2].to_i..(match[3] || match[2]).to_i)] if match
+      end
+      extraction.lines.filter_map do |line|
+        location = /\Apage (\d+), line (\d+)\z/.match(line.location)
+        next if location && item_spans.any? { |page, span| page == location[1] && span.cover?(location[2].to_i) }
+
+        text = line.text
+        next if DEMAND.match?(text) || BOILERPLATE.match?(text)
+        next unless text.match?(/\b(?:MTRs?|COA|DFARS|REACH|RoHS|mill test report|certificate of analysis)\b/i)
+        next unless text.match?(/\b(?:required|require|all|each|must|compliant)\b/i)
+
+        text
+      end.join(" ")
+    end
+
+    def self.table_row?(line)
+      raw = line.layout_text || line.text
+      # At least two horizontal cells: a description and a quantity/unit. The
+      # table header supplies context, so unknown product names remain visible.
+      return false if BOILERPLATE.match?(raw)
+
+      cells = raw.strip.split(/\s{2,}/)
+      return false if cells.length < 2
+
+      # A demand must occupy table cells, not just occur inside a description
+      # such as "CONTAINER SIZE: 1000 KG PER BOX".
+      cells.each_with_index.any? do |cell, index|
+        next false if index.zero?
+
+        cell.match?(/\A(?:#{DEMAND})\z/) ||
+          (cell.match?(/\A#{UNIT}\z/i) && [cells[index - 1], cells[index + 1]].compact.any? { |value| value.match?(/\A#{NUMBER}\z/) })
+      end
+    end
+
+    def self.without_line_number(line)
+      # Only a table explicitly headed "Line" establishes this as a row number.
+      # Keep product identifiers in tables headed "Item" intact.
+      Extractor::Line.new(text: line.text.sub(/\A\d+\s{2,}/, ""), location: line.location, layout_text: line.layout_text)
+    end
+
+    def self.assemble(lines)
+      first = lines.first
+      last = lines.last
+      location = first.location
+      if lines.length > 1 && first.location.match?(/, line \d+\z/) && last.location.match?(/, line \d+\z/)
+        location = first.location.sub(/, line (\d+)\z/, ", lines \\1-#{last.location[/\d+\z/]}")
+      end
+      Extractor::Line.new(text: lines.map(&:text).join(" "), location: location)
+    end
+  end
+end

+module RfqProcessing
+  # A small chat completions client for OpenAI-compatible gateways: the hackathon's
+  # Nicrron gateway in front of Claude. Standard library only, Net::HTTP and JSON.
+  #
+  # One call is one POST to <base_url>/chat/completions with a strict json_schema
+  # response_format. A 400/422 rejecting the schema format is retried
+  # once without response_format, with the schema
+  # written into the system message. A 502, 503, or 504 is read as a flaky gateway:
+  # the same request is sent once more after a short pause. Nothing here raises;
+  # every failure is one warn line and nil, so the pipeline keeps the rules reading.
+  # The API key travels in the Authorization header and is never written to a log line.
+  class OpenAiCompatibleClient
+    TRANSIENT_STATUSES = [502, 503, 504].freeze
+    TRANSIENT_PAUSE_SECONDS = 1
+    SCHEMA_NAME = "rfq_line"
+    PLAIN_JSON_INSTRUCTION = "Answer with exactly one JSON object and nothing else, matching this JSON schema:"
+
+    attr_reader :base_url, :model, :last_status
+
+    # http: a callable for tests, ->(request) { [status, body_string] }, where request is a Hash
+    # with string keys "method" ("POST" or "GET"), "url", "headers", and "body" (the body as a
+    # Hash, nil for a GET). Nil means Net::HTTP.
+    def initialize(api_key:, base_url:, model:, open_timeout: 10, read_timeout: 90, http: nil)
+      @api_key = api_key.to_s
+      @base_url = base_url.to_s.strip.sub(%r{/+\z}, "")
+      @model = model.to_s
+      @open_timeout = open_timeout
+      @read_timeout = read_timeout
+      @http = http || method(:over_net_http)
+      @last_status = nil
+    end
+
+    def url
+      "#{@base_url}/chat/completions"
+    end
+
+    def models_url
+      "#{@base_url}/models"
+    end
+
+    # { "text" => the assistant message content, "model" => the model the gateway names, or nil }.
+    # Nil on any failure, after one warn line.
+    def chat(system:, user:, schema:, max_tokens:)
+      status, raw = post(structured_body(system, user, schema, max_tokens))
+      return nil if status.nil?
+
+      if schema_rejected?(status, raw)
+        first = status
+        status, raw = post(plain_body(system, user, schema, max_tokens))
+        return nil if status.nil?
+
+        unless success?(status)
+          warn "OpenAiCompatibleClient: #{where(url)} answered #{first} with response_format, then #{status} without it, for model #{@model}; no reading"
+          return nil
+        end
+        warn "OpenAiCompatibleClient: #{where(url)} rejected response_format (#{first}); the retry with the schema in the system message succeeded"
+      elsif !success?(status)
+        warn "OpenAiCompatibleClient: #{where(url)} answered #{status} for model #{@model}: #{snippet(raw)}"
+        return nil
+      end
+      extract(raw)
+    rescue StandardError => e
+      warn "OpenAiCompatibleClient: request to #{where(url)} failed (#{e.class})"
+      nil
+    end
+
+    # The model ids the gateway lists at GET <base_url>/models (the OpenAI shape, {"data" => [{"id" => ...}]},
+    # or a bare array). A diagnostic: one request, no retry, so a flaky answer shows as it is.
+    # Nil after one warn line when the answer is not 2xx, the request fails, or the body has no list.
+    def models
+      status, raw = send_once("GET", models_url, nil)
+      unless success?(status)
+        warn "OpenAiCompatibleClient: GET #{where(models_url)} answered #{status}: #{snippet(raw)}"
+        return nil
+      end
+      parsed = JSON.parse(raw)
+      list = parsed.is_a?(Hash) ? parsed["data"] : parsed
+      unless list.is_a?(Array)
+        warn "OpenAiCompatibleClient: GET #{where(models_url)} answered 2xx with no model list: #{snippet(raw)}"
+        return nil
+      end
+      list.filter_map { |entry| entry.is_a?(Hash) ? entry["id"] : entry }.map(&:to_s).reject(&:empty?)
+    rescue StandardError => e
+      warn "OpenAiCompatibleClient: GET #{where(models_url)} failed (#{e.class})"
+      nil
+    end
+
+    private
+
+    def schema_rejected?(status, raw)
+      [400, 422].include?(status) && raw.to_s.match?(/response_format|json_schema/i)
+    end
+
+    # [status, body] after at most one same-request retry on a transient status.
+    # Nil when the retry is transient too, after the one warn line.
+    def post(body)
+      status, raw = send_once("POST", url, body)
+      return [status, raw] unless TRANSIENT_STATUSES.include?(status)
+
+      sleep(TRANSIENT_PAUSE_SECONDS)
+      status, raw = send_once("POST", url, body)
+      return [status, raw] unless TRANSIENT_STATUSES.include?(status)
+
+      warn "OpenAiCompatibleClient: #{where(url)} answered #{status} twice for model #{@model}; giving up"
+      nil
+    end
+
+    def send_once(method, target, body)
+      status, raw = @http.call({ "method" => method, "url" => target, "headers" => headers, "body" => body })
+      @last_status = Integer(status)
+      [@last_status, raw.to_s]
+    end
+
+    def headers
+      { "Authorization" => "Bearer #{@api_key}", "Content-Type" => "application/json" }
+    end
+
+    def structured_body(system, user, schema, max_tokens)
+      base_body(system, user, max_tokens).merge(
+        "response_format" => { "type" => "json_schema",
+                               "json_schema" => { "name" => SCHEMA_NAME, "strict" => true, "schema" => schema } }
+      )
+    end
+
+    def plain_body(system, user, schema, max_tokens)
+      base_body("#{system.to_s.rstrip}\n\n#{PLAIN_JSON_INSTRUCTION}\n#{JSON.generate(schema)}", user, max_tokens)
+    end
+
+    def base_body(system, user, max_tokens)
+      { "model" => @model,
+        "messages" => [{ "role" => "system", "content" => system.to_s }, { "role" => "user", "content" => user.to_s }],
+        "max_tokens" => max_tokens,
+        "temperature" => 0 }
+    end
+
+    def extract(raw)
+      parsed = JSON.parse(raw)
+      choice = parsed.is_a?(Hash) && parsed["choices"].is_a?(Array) ? parsed["choices"][0] : nil
+      message = choice.is_a?(Hash) ? choice["message"] : nil
+      content = message.is_a?(Hash) ? message["content"] : nil
+      content = content.map { |part| part.is_a?(Hash) ? part["text"].to_s : part.to_s }.join if content.is_a?(Array)
+      if !content.is_a?(String) || content.strip.empty?
+        warn "OpenAiCompatibleClient: #{where(url)} answered with no choices[0].message.content"
+        return nil
+      end
+      { "text" => content, "model" => (parsed["model"].is_a?(String) ? parsed["model"] : nil) }
+    rescue JSON::ParserError
+      warn "OpenAiCompatibleClient: #{where(url)} answered 2xx with a body that is not JSON: #{snippet(raw)}"
+      nil
+    end
+
+    def success?(status)
+      (200..299).cover?(status)
+    end
+
+    # Host and path only: no query, no credentials, safe for a log line.
+    def where(target)
+      uri = URI.parse(target)
+      "#{uri.host}#{uri.path}"
+    rescue URI::InvalidURIError
+      "invalid gateway URL"
+    end
+
+    def snippet(raw)
+      text = raw.to_s
+      text = text.gsub(@api_key, "[redacted]") unless @api_key.empty?
+      text.gsub(/\s+/, " ").strip[0, 200]
+    end
+
+    def over_net_http(request)
+      uri = URI.parse(request["url"])
+      http = Net::HTTP.new(uri.host, uri.port)
+      http.use_ssl = uri.scheme == "https"
+      http.open_timeout = @open_timeout
+      http.read_timeout = @read_timeout
+      call = if request["method"] == "GET"
+               Net::HTTP::Get.new(uri.request_uri, request["headers"])
+             else
+               post = Net::HTTP::Post.new(uri.request_uri, request["headers"])
+               post.body = JSON.generate(request["body"])
+               post
+             end
+      response = http.start { |connection| connection.request(call) }
+      [response.code.to_i, response.body.to_s]
+    end
